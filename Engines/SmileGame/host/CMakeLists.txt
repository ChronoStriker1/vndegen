cmake_minimum_required(VERSION 3.24)
project(SmileGameHost C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

set(RUNTIME ${CMAKE_CURRENT_SOURCE_DIR}/../../../Packages/VNDegenKit/Sources/CSmileGame)

file(GLOB RUNTIME_SOURCES ${RUNTIME}/*.c)
add_library(smilegame STATIC ${RUNTIME_SOURCES})
target_include_directories(smilegame PUBLIC ${RUNTIME} ${RUNTIME}/include)
target_compile_options(smilegame PRIVATE -O2 -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare -Wno-unused-function)
set_source_files_properties(${RUNTIME}/ufbx_unit.c PROPERTIES COMPILE_OPTIONS "-w")
target_compile_definitions(smilegame PRIVATE _GNU_SOURCE)
target_link_libraries(smilegame PUBLIC m)

add_executable(sgrun sgrun.c)
target_link_libraries(sgrun PRIVATE smilegame)

enable_testing()
add_executable(sgtest sgtest.c)
target_link_libraries(sgtest PRIVATE smilegame)
add_test(NAME formats COMMAND sgtest)
add_executable(sgtest_B sgtest_B.c)
target_link_libraries(sgtest_B PRIVATE smilegame)
add_test(NAME tables_B COMMAND sgtest_B)
add_executable(sgtest_E sgtest_E.c)
target_link_libraries(sgtest_E PRIVATE smilegame)
add_test(NAME saves_audio COMMAND sgtest_E)
add_executable(sgtest_D sgtest_D.c)
target_link_libraries(sgtest_D PRIVATE smilegame)
add_test(NAME render COMMAND sgtest_D)
add_executable(sgview sgview.c)
target_link_libraries(sgview PRIVATE smilegame)
include(sgtest_A.cmake)
add_executable(sgtest_C sgtest_C.c)
target_link_libraries(sgtest_C PRIVATE smilegame)
add_test(NAME battle_C COMMAND sgtest_C)
add_executable(sgbattle sgbattle.c)
target_link_libraries(sgbattle PRIVATE smilegame)

find_package(SDL2 CONFIG)
if(SDL2_FOUND)
  add_executable(sgplay sgplay.c)
  target_link_libraries(sgplay PRIVATE smilegame SDL2::SDL2)
  add_executable(sgbattleplay sgbattle.c)
  target_compile_definitions(sgbattleplay PRIVATE SG_SDL)
  target_link_libraries(sgbattleplay PRIVATE smilegame SDL2::SDL2)
endif()
add_executable(sgtest_G sgtest_G.c)
target_link_libraries(sgtest_G PRIVATE smilegame)
add_test(NAME wiring COMMAND sgtest_G)
include(sgtest_H.cmake)
add_executable(sgtest_I sgtest_I.c)
target_link_libraries(sgtest_I PRIVATE smilegame)
add_test(NAME field_I COMMAND sgtest_I)
add_executable(sgtest_J sgtest_J.c)
target_compile_definitions(sgtest_J PRIVATE SG_FONTS="${CMAKE_CURRENT_SOURCE_DIR}/../../Shared/fonts")
target_link_libraries(sgtest_J PRIVATE smilegame)
add_test(NAME release_J COMMAND sgtest_J)
add_executable(sgtest_L sgtest_L.c)
target_link_libraries(sgtest_L PRIVATE smilegame)
add_test(NAME colette_L COMMAND sgtest_L)
